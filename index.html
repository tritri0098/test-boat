src/main.js 
// Ajouter un bateau GLB
const boat = await world.addBoatFromUrl({
  id: "rescue",
  url: new URL("../assets/models/rescue.glb", import.meta.url).href,
  position: [5, 0, -10],
  scale: 0.8
});

// Déplacer la caméra
world.cameraController.setPose({
  position: [4, 2.5, 8],
  target: [0, 0.8, -5]
});

// Ajouter une scène animée
slides.addScene({
  id: "heavy-storm",
  weather: { storm: 1, rain: 1, wind: 24 },
  camera: { position: [2, 1.6, 7], target: [0, 0.8, -5] }
});
